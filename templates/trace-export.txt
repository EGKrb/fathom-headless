// Records an execution trace from the entry point and exports it to JSON, CSV, Tenet and .ftt (Time Travel).
// Needs an x64dbg build with the TraceExport command.
// Change the stop condition (0 = never) and the maximum count (.N = decimal) as needed,
// for example: TraceIntoConditional "cip == kernelbase.ExitProcess", .1000000

settingset Events, SystemBreakpoint, 0
settingset Events, TlsCallbacks, 0
settingset Events, EntryBreakpoint, 1
init "%TARGET%"

StartTraceRecording "%OUT%\trace.trace%BITS%"
// Stops at the process exit too, so that a short-lived target can still be exported
TraceIntoConditional "cip == ntdll.RtlExitUserProcess", .50000
StopTraceRecording

TraceExport "%OUT%\trace.trace%BITS%", "%OUT%\trace.json"
TraceExport "%OUT%\trace.trace%BITS%", "%OUT%\trace.csv"
// Tenet text trace for IDA Pro; on x64, a Frinet one can be added with: TraceExport "...", "%OUT%\trace.tenet", frinet
TraceExport "%OUT%\trace.trace%BITS%", "%OUT%\trace.log", tenet
// Indexed, compressed trace for the tt.* commands (ExtraCmds) and the Time Travel tab of Fathom;
// regs=all adds the x87 / SSE / AVX registers
TraceExport "%OUT%\trace.trace%BITS%", "%OUT%\trace.ftt", ftt
log "[template] done"
